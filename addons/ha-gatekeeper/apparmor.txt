#include <tunables/global>

# Enforce mode. Supervisor rewrites the top-level name on installation.
profile ha_gatekeeper flags=(attach_disconnected,mediate_deleted) {
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  capability chown,
  capability fowner,
  capability setuid,
  capability setgid,
  capability setpcap,
  signal (send, receive) set=(term, kill, int, chld),
  # Go self-thread preemption requires SIGURG; match only the stacked connector.
  signal (send, receive) set=(urg) peer="**//&**//cloudflared",
  deny ptrace (read, trace),
  /run.sh rix,
  /run-user.sh rix,
  /{usr/,}bin/{sh,dash} rix,
  /usr/bin/setpriv rix,
  /usr/bin/env rix, # Script shebang interpreter.
  # Prisma platform detection fallback helpers (no arbitrary binary execution).
  /usr/bin/{uname,openssl,sed,grep,sort} rix,
  /usr/sbin/ldconfig rix,
  /usr/local/bin/node mrix,
  /usr/local/bin/{npm,npx} rix,
  /usr/local/lib/node_modules/** r,
  /usr/local/lib/node_modules/npm/bin/{npm,npx}-cli.js rix,
  /app/ r,
  /app/** r,
  /app/**/node_modules/**.so{,.*} mr,
  /app/{,**/}node_modules/@prisma/engines/schema-engine-* rix,
  /app/{,**/}node_modules/prisma/**/schema-engine-* rix,
  /{usr/,}lib/** mr,
  /etc/{ld.so.cache,ld.so.conf,ld.so.conf.d/**,nsswitch.conf,passwd,group,hosts,hostname,resolv.conf,gai.conf,localtime,timezone,os-release} r,
  /etc/ssl/** r,
  /usr/share/{zoneinfo,ca-certificates}/** r,
  /dev/{null,zero,random,urandom} rw,
  /proc/ r,
  /proc/** r,
  /sys/devices/system/cpu/** r,
  /sys/fs/cgroup/** r,
  deny /proc/**/{environ,mem} r,
  # Proc writes are default-denied except the own AppArmor attribute below.
  deny /sys/** w,
  /data/ rw,
  /data/options.json rw,
  # Root bootstrap prepares legacy UID-1000 files; runtime uses DAC mode 0600.
  /data/{secrets.json,ha-gatekeeper.db,ha-gatekeeper.db-journal,ha-gatekeeper.db-wal,ha-gatekeeper.db-shm} rwk,
  /tmp/ r,
  owner /tmp/ha-gatekeeper-*/ rw,
  owner /tmp/ha-gatekeeper-*/** rwk,
  # The native launcher resolves Supervisor's installed name and requests a
  # kernel STACK of this profile's child before exec. No named-exec fallback.
  /usr/local/bin/cloudflared mrix,
  /usr/local/libexec/cloudflared mrix,
  owner /proc/[0-9]*/attr/apparmor/current w,
  change_profile -> &{ha_gatekeeper,*_ha_gatekeeper}//cloudflared,

  # No application, shell, /data or parent memory access in the connector.
  profile cloudflared flags=(attach_disconnected,mediate_deleted) {
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,
    signal (receive) set=(term, kill, int),
    signal (send) set=(chld),
    signal (send, receive) set=(urg) peer="**//&**//cloudflared",
    deny ptrace (read, trace),
    /usr/local/libexec/cloudflared mrix,
    deny /usr/local/bin/cloudflared x,
    /{usr/,}lib/** mr,
    /etc/{ld.so.cache,nsswitch.conf,hosts,hostname,resolv.conf,gai.conf,localtime,timezone,os-release} r,
    /etc/ssl/** r,
    /usr/share/{zoneinfo,ca-certificates}/** r,
    /dev/{null,zero,random,urandom} rw,
    /proc/ r,
    /proc/** r,
    /sys/devices/system/cpu/** r,
    /sys/fs/cgroup/** r,
    deny /proc/**/{environ,mem} r,
    deny /proc/** w,
    deny /sys/** w,
    deny /data{,/**} rwklmx,
    deny /app{,/**} rwklmx,
    deny /{usr/,}bin/** x,
    deny /usr/local/bin/node x,
    owner /tmp/ha-gatekeeper-*/hgk-cloudflared-*/ r,
    owner /tmp/ha-gatekeeper-*/hgk-cloudflared-*/token r,
  }
}
